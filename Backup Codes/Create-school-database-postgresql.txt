    /// <summary>
    /// API endpoint to handle the creation of new school databases.
    /// This endpoint MUST be secured to prevent unauthorized access.
    /// </summary>
    [ApiController]
    [Route("api/[controller]")]
    // 🔐 CRITICAL: Protect this endpoint! Only Super Admins should be able to provision databases.
    // [Authorize(Policy = "SuperAdminOnly")] 
    public class SchoolSystemsController : ControllerBase
    {
        private readonly ILogger<SchoolSystemsController> _logger;
        private readonly IDatabaseProvisioningService _provisioningService;

        public SchoolSystemsController(IDatabaseProvisioningService provisioningService, ILogger<SchoolSystemsController> logger)
        {
            _provisioningService = provisioningService;
            _logger = logger;
        }

        /// <summary>
        /// Receives database details and provisions a new PostgreSQL database and schema.
        /// </summary>
        [HttpPost("provision")]
        [ProducesResponseType(typeof(ProvisionResponseModel), 200)]
        [ProducesResponseType(typeof(string), 400)]
        [ProducesResponseType(typeof(string), 500)]
        public async Task<IActionResult> ProvisionDatabase([FromBody] ProvisionRequestModel model)
        {
            try
            {
                var connectionString = await _provisioningService.ProvisionNewSchoolDatabaseAsync(model);
                
                _logger.LogInformation("Successfully provisioned database '{DbName}' on server {ServerIP}", model.NewDbName, model.ServerIP);

                // You would now save this connectionString and other details
                // to your master 'SystemDetails' table.

                return Ok(new ProvisionResponseModel { 
                    Message = $"Database '{model.NewDbName}' provisioned successfully.", 
                    ConnectionString = connectionString 
                });
            }
            catch (ArgumentException ex)
            {
                _logger.LogWarning("Invalid provision request: {Message}", ex.Message);
                return BadRequest(ex.Message);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "An unexpected error occurred during database provisioning for '{DbName}'.", model.NewDbName);
                return StatusCode(500, $"An internal error occurred: {ex.Message}");
            }
        }
    }

    #endregion

    #region "2. Service Layer - The Core Logic"

    public interface IDatabaseProvisioningService
    {
        /// <summary>
        /// Creates a new PostgreSQL database and populates it with an initial schema.
        /// </summary>
        /// <param name="model">The details for the new database connection.</param>
        /// <returns>The connection string for the newly created database.</returns>
        Task<string> ProvisionNewSchoolDatabaseAsync(ProvisionRequestModel model);
    }

    public class DatabaseProvisioningService : IDatabaseProvisioningService
    {
        public async Task<string> ProvisionNewSchoolDatabaseAsync(ProvisionRequestModel model)
        {
            // --- Step 1: Validate Input ---
            ValidateAndSanitizeModel(model);

            // --- Step 2: Build Connection Strings ---
            var maintenanceDbConnectionString = new NpgsqlConnectionStringBuilder
            {
                Host = model.ServerIP,
                Port = model.ServerPort,
                Database = "postgres", // Connect to the maintenance DB to run CREATE DATABASE
                Username = model.DbUsername,
                Password = model.DbPassword,
                SslMode = SslMode.Prefer // Or Require, depending on your server config
            }.ConnectionString;

            var newDbConnectionString = new NpgsqlConnectionStringBuilder
            {
                Host = model.ServerIP,
                Port = model.ServerPort,
                Database = model.NewDbName, // The final connection string for the new DB
                Username = model.DbUsername,
                Password = model.DbPassword,
                SslMode = SslMode.Prefer
            }.ConnectionString;

            // --- Step 3: Create the Database ---
            await using (var connection = new NpgsqlConnection(maintenanceDbConnectionString))
            {
                await connection.OpenAsync();
                
                // Check if the database already exists
                var checkCmdText = $"SELECT 1 FROM pg_database WHERE datname = @dbName";
                await using (var checkCmd = new NpgsqlCommand(checkCmdText, connection))
                {
                    checkCmd.Parameters.AddWithValue("dbName", model.NewDbName);
                    var exists = await checkCmd.ExecuteScalarAsync();
                    if (exists != null)
                    {
                        // Database already exists, you can decide to throw an error or continue
                        throw new ArgumentException($"Database '{model.NewDbName}' already exists.");
                    }
                }

                // Create the database. Using a simple command as the DB name is already sanitized.
                var createCmdText = $"CREATE DATABASE \"{model.NewDbName}\"";
                await using (var createCmd = new NpgsqlCommand(createCmdText, connection))
                {
                    await createCmd.ExecuteNonQueryAsync();
                }
            }

            // --- Step 4: Use EF Core to Create Schema (Tables) ---
            var optionsBuilder = new DbContextOptionsBuilder<SchoolDbContext>();
            optionsBuilder.UseNpgsql(newDbConnectionString);

            await using (var context = new SchoolDbContext(optionsBuilder.Options))
            {
                // This command connects to the new DB and runs CREATE TABLE scripts
                // based on your DbContext model.
                await context.Database.EnsureCreatedAsync();
            }

            return newDbConnectionString;
        }

        private void ValidateAndSanitizeModel(ProvisionRequestModel model)
        {
            // Basic validation
            if (string.IsNullOrWhiteSpace(model.NewDbName) || string.IsNullOrWhiteSpace(model.DbUsername))
            {
                throw new ArgumentException("Database name and username are required.");
            }

            // 🛡️ Security: Sanitize database name to prevent SQL injection.
            // Only allow alphanumeric characters and underscores.
            if (!Regex.IsMatch(model.NewDbName, @"^[a-zA-Z0-9_]+$"))
            {
                throw new ArgumentException("Invalid database name. Only letters, numbers, and underscores are allowed.");
            }
        }
    }


    #endregion
